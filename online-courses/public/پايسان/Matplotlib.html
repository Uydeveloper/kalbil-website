كىرىسۆز Matplotlib Seaborn Bokeh Plotly سېلىشتۇرۇش دېمو

سانلىق مەلۇمات كۆرسىتىش ھەققىدە

بۈگۈنكى كۈندە، كۈندىلىك تۇرمۇشتا نۇرغۇن سانلىق مەلۇماتلار بارلىققا كېلىدۇ. بۇ سانلىق مەلۇماتلارنى ئىلمىي تەھلىل قىلىش ئۈچۈن ئۇلارنىڭ خام شەكلىدىكى ئالاقىسىزلىقىنى چۈشىنىش تەس بولۇشى مۇمكىن. بۇ مەسىلىنى ھەل قىلىش ئۈچۈن سانلىق مەلۇمات كۆرسىتىش ئىشلىتىلىدۇ. سانلىق مەلۇمات كۆرسىتىش سانلىق مەلۇماتلارنى چۈشىنىش، كۆزىتىش ۋە تەھلىل قىلىشنى ئاسانلاشتۇرىدىغان ياخشى تەشكىللەنگەن رەسىملىك كۆرسىتىشنى تەمىنلەيدۇ.

بۇ دەرسلىكتە، بىز Python نىڭ تۆرلۈك كۈتۇپخانىلىرى ئارقىلىق سانلىق مەلۇماتلارنى قانداق كۆرسىتىش توغرىسىدا مۇزاكىرە قىلىمىز.

ئىشلىتىلگەن سانلىق مەلۇمات بازىسى

Matplotlib

Matplotlib ئاسان ئىشلىتىشچان، تۆۋەن سەۋىيىلىك سانلىق مەلۇمات كۆرسىتىش كۈتۇپخانىسى بولۇپ، NumPy تىزىملىكلىرى ئۈستىدە قۇرۇلغان. ئۇ تارقاق نۇقتا گرافىكى، سىزىق گرافىكى، ستون گرافىكى قاتارلىق تۈرلۈك گرافىكىلارنى ئۆز ئىچىگە ئالىدۇ. Matplotlib نۇرغۇن ئىشلىنىشچانلىقنى تەمىنلەيدۇ.

Matplotlib نى ئورنىتىش 
pip install matplotlib

تارقاق نۇقتا گرافىكى (Scatter Plot)

تارقاق نۇقتا گرافىكى ئۆزگەرگۈچىلەر ئارىسىدىكى مۇناسىۋەتنى كۆزىتىش ئۈچۈن ئىشلىتىلىدۇ ۋە ئۇلار ئارىسىدىكى مۇناسىۋەتنى كۆرسىتىش ئۈچۈن نۇقتىلارنى ئىشلىتىدۇ. matplotlib كۈتۇپخانىسىدىكى scatter() ئۇسۇلى تارقاق نۇقتا گرافىكىنى سىزىش ئۈچۈن ئىشلىتىلىدۇ.

ئاددىي تارقاق نۇقتا گرافىكى 
import pandas as pd
import matplotlib.pyplot as plt

# سانلىق مەلۇمات بازىسىنى ئوقۇش
data = pd.read_csv("tips.csv")

# كۈنگە قارشى چەك ئۈچۈن تارقاق نۇقتا گرافىكى
plt.scatter(data['day'], data['tip'])

# گرافىكىغا ماۋزۇ قوشۇش
plt.title("تارقاق نۇقتا گرافىكى")

# X ۋە Y بەلگىلىرىنى تەڭشەش
plt.xlabel('كۈن')
plt.ylabel('چەك')

plt.show()
رەڭ ۋە چوڭلۇق بىلەن تارقاق نۇقتا گرافىكى 
import pandas as pd
import matplotlib.pyplot as plt

# سانلىق مەلۇمات بازىسىنى ئوقۇش

            s=data['total_bill'])

plt.title("تارقاق نۇقتا گرافىكى")

plt.ylabel('چەك')

plt.colorbar()

سىزىق گرافىكى (Line Chart)

سىزىق گرافىكى ئوخشىمىغان ئوكۇلاردىكى ئىككى سانلىق مەلۇمات X ۋە Y ئارىسىدىكى مۇناسىۋەتنى كۆرسىتىش ئۈچۈن ئىشلىتىلىدۇ. ئۇ plot() فۇنكسىيەسى ئارقىلىق سىزىلىدۇ.

سىزىق گرافىكى 
import pandas as pd
import matplotlib.pyplot as plt

# سانلىق مەلۇمات بازىسىنى ئوقۇش
data = pd.read_csv("tips.csv")

# چەككە قارشى كۈن ئۈچۈن تارقاق نۇقتا گرافىكى
plt.plot(data['tip'])
plt.plot(data['size'])

# گرافىكىغا ماۋزۇ قوشۇش
plt.title("سىزىق گرافىكى")

# X ۋە Y بەلگىلىرىنى تەڭشەش
plt.xlabel('كۈن')
plt.ylabel('چەك')

plt.show()

ستون گرافىكى (Bar Chart)

ستون گرافىكى ياكى ستون دىئاگراممىسى ئۇلار كۆرسىتىدىغان قىممەتلەرگە مۇناسىۋەتلىك ئۇزۇنلۇق ۋە ئېگىزلىك بىلەن سانلىق مەلۇمات تۈرىنى تەمىنلەيدىغان تىك تۆت تەرەپلىك ستونلار بىلەن ۋەكىللىك قىلىنغان گرافىكتۇر. ئۇ bar() ئۇسۇلى ئارقىلىق قۇرۇلالايدۇ.

ستون گرافىكى 
import pandas as pd
import matplotlib.pyplot as plt

# سانلىق مەلۇمات بازىسىنى ئوقۇش
data = pd.read_csv("tips.csv")

# چەككە قارشى كۈن ئۈچۈن ستون گرافىكى
plt.bar(data['day'], data['tip'])

plt.title("ستون گرافىكى")

# X ۋە Y بەلگىلىرىنى تەڭشەش
plt.xlabel('كۈن')
plt.ylabel('چەك')

plt.show()

ھىستوگرامما (Histogram)

ھىستوگرامما ئاساسەن بەزى گۇرۇپپىلار شەكلىدىكى سانلىق مەلۇماتلارنى كۆرسىتىش ئۈچۈن ئىشلىتىلىدۇ. ئۇ X-ئوك بىن دائىرىسىنى ۋەكىللىك قىلغاندا، Y-ئوك يىغىندىق سانى ھەققىدە ئۇچۇر بېرىدىغان ستون دىئاگراممىسىنىڭ بىر تۈرىدۇر. hist() فۇنكسىيەسى ھىستوگرامما ھېسابلاش ۋە قۇرۇش ئۈچۈن ئىشلىتىلىدۇ.

ھىستوگرامما 
import pandas as pd
import matplotlib.pyplot as plt

# سانلىق مەلۇمات بازىسىنى ئوقۇش
data = pd.read_csv("tips.csv")

# total_bills نىڭ ھىستوگراممىسى
plt.hist(data['total_bill'])

plt.title("ھىستوگرامما")

plt.show()

Seaborn

Seaborn ماتپلوتلىبنىڭ ئۈستىگە قۇرۇلغان يۇقىرى سەۋىيە ئارالمىسىدۇر. ئۇ تېخىمۇ جەلپكار گرافىكىلارنى قۇرۇش ئۈچۈن گۈزەل دىزايىن ئۇسلۇبلىرى ۋە رەڭ پالېتلىرىنى تەمىنلەيدۇ.

ئورنىتىش

Seaborn نى ئورنىتىش 
pip install seaborn

Seaborn ماتپلوتلىبنىڭ ئۈستىگە قۇرۇلغان بولغاچقا، ئۇ ماتپلوتلىب بىلەن بىرلىكتە ئىشلىتىلەلەيدۇ. ماتپلوتلىب ۋە Seaborn نى بىرلىكتە ئىشلىتىش بەك ئاددىي جەرياندۇر. بىز ئادەتتىكىدەك Seaborn سىزىش فۇنكسىيەسىنى چاقىرىشىمىز كېرەك، ئاندىن ماتپلوتلىبنىڭ ماسلاشتۇرۇش فۇنكسىيەسىنى ئىشلىتەلەيمىز.

Seaborn ۋە Matplotlib نى بىرلىكتە ئىشلىتىش 
# كۈتۇپخانىلارنى ئەكىرەش
import seaborn as sns
import matplotlib.pyplot as plt
import pandas as pd

# سانلىق مەلۇمات بازىسىنى ئوقۇش
data = pd.read_csv("tips.csv")

# سىزىق گرافىكىسىنى سىزىش
sns.lineplot(x="sex", y="total_bill", data=data)

# ماتپلوتلىب فۇنكسىيەسى ئارقىلىق ماۋزۇنى تەڭشەش
plt.title('ماتپلوتلىب فۇنكسىيەسى ئارقىلىق ماۋزۇ')

plt.show()

Bokeh

بىزنىڭ تىزىملىكىمىزنىڭ ئۈچىنچى كۈتۇپخانىسىغا يۆتكىلىمىز. Bokeh ئاساسەن ئۇنىڭ ئىنتېراكتىپ گرافىكا كۆرسىتىشى ئۈچۈن مەشھۇردۇر. Bokeh زامانىۋى تور كۆرگۈچلەرنى زىننەتلىك، قىسقا قۇرۇلمىلىق يۇقىرى سەۋىيەلىك ئىنتېراكتىپلىق بىلەن تەقدىم قىلىش ئۈچۈن HTML ۋە JavaScript نى ئىشلىتىپ گرافىكىلىرىنى رەنده قىلىدۇ.

ئورنىتىش

Bokeh نى ئورنىتىش 
pip install bokeh

تارقاق نۇقتا گرافىكى (Scatter Plot)

Bokeh دىكى تارقاق نۇقتا گرافىكى plotting مودۇلىنىڭ scatter() ئۇسۇلى ئارقىلىق سىزىلالايدۇ. بۇ يەردە ماس ھالدا x ۋە y كووردىناتلىرىنى بېرىڭ.

تارقاق نۇقتا گرافىكى 
# مودۇللارنى ئەكىرەش
from bokeh.plotting import figure, output_file, show
from bokeh.palettes import magma
import pandas as pd

# شەكىل نەرسىسىنى مىسال قىلىش
graph = figure(title = "Bokeh تارقاق نۇقتا گرافىكى")

# سانلىق مەلۇمات بازىسىنى ئوقۇش
data = pd.read_csv("tips.csv")

color = magma(256)

# گرافىكىنى سىزىش
graph.scatter(data['total_bill'], data['tip'], color=color)

# مودېلنى كۆرسىتىش
show(graph)

ئىنتېراكتىپ سانلىق مەلۇمات كۆرسىتىش

Bokeh نىڭ ئاساسلىق ئالاھىدىلىكى گرافىكىلارغا ئىنتېراكتىپلىق قوشۇش. بىز قوشۇلالايدىغان تۈرلۈك ئىنتېراكتىپلىقلارنى كۆرۈپ چىقايلى.

ئىنتېراكتىپ ئۇلانما (Legend)

click_policy خاسلىقى ئۇلانمىنى ئىنتېراكتىپ قىلىدۇ. ئىككى خىل ئىنتېراكتىپلىق بار:

  • يوشۇرۇش: گلىفلارنى يوشۇرىدۇ.
  • ئاۋازسىز: گلىفنى يوشۇرۇش ئۇنى پۈتۈنلەي يوق قىلىدۇ، ئەمما گلىفنى ئاۋازسىز قىلىش گلىفنى پارامېترلارغا ئاساسەن پەقەت ئاستىلىققا چۈشۈرىدۇ.
ئىنتېراكتىپ ئۇلانما 
# مودۇللارنى ئەكىرەش
from bokeh.plotting import figure, output_file, show
import pandas as pd

# شەكىل نەرسىسىنى مىسال قىلىش
graph = figure(title = "Bokeh ستون گرافىكى")

# سانلىق مەلۇمات بازىسىنى ئوقۇش
data = pd.read_csv("tips.csv")

# گرافىكىنى سىزىش
graph.vbar(data['total_bill'], top=data['tip'], 
           legend_label = "ھېساب چەككە قارشى", color='green')

graph.vbar(data['tip'], top=data['size'], 
           legend_label = "چەك ئۆلچەمگە قارشى", color='red')

graph.legend.click_policy = "hide"

# مودېلنى كۆرسىتىش
show(graph)

ۋىدجېتلارنى قوشۇش

Bokeh توپچا، سىزغۇچ، تەكشۈرۈش رامكىسى قاتارلىق HTML جەدۋەللىرىگە ئوخشاش GUI ئالاھىدىلىكلىرىنى تەمىنلەيدۇ. بۇلار گرافىكىنىڭ پارامېترلىرىنى ئۆزگەرتىش، گرافىكا سانلىق مەلۇماتلىرىنى ماسلاشتۇرۇش قاتارلىقلارغا يول قويىدىغان گرافىكىغا ئىنتېراكتىپ ئارالمىنى تەمىنلەيدۇ. ئادەتتىكى ئىشلىتىلىدىغان ۋىدجېتلارنى قانداق ئىشلىتىش ۋە قوشۇشنى كۆرۈپ چىقايلى.

ۋىدجېتلارنى قوشۇش 
from bokeh.io import show
from bokeh.models import Button, CheckboxGroup, RadioGroup, CustomJS

button = Button(label="GFG")

button.js_on_click(CustomJS(
    code="console.log('button: click!', this.toString())"))

# تەكشۈرۈش رامكىسى ۋە راديو توپچىسى ئۈچۈن بەلگىلەر
L = ["بىرىنچى", "ئىككىنچى", "ئۈچىنچى"]

# active پارامېتىرى سەيلەنگەن قىممەتنى سەكسىز تەڭشەيدۇ
checkbox_group = CheckboxGroup(labels=L, active=[0, 2])

checkbox_group.js_on_click(CustomJS(code="""
    console.log('checkbox_group: active=' + this.active, this.toString())
"""))

# active پارامېتىرى سەيلەنگەن قىممەتنى سەكسىز تەڭشەيدۇ
radio_group = RadioGroup(labels=L, active=1)

radio_group.js_on_click(CustomJS(code="""
    console.log('radio_group: active=' + this.active, this.toString())
"""))

show(button)
show(checkbox_group)
show(radio_group)

Plotly

بۇ بىزنىڭ تىزىملىكىمىزنىڭ ئاخىرقى كۈتۇپخانىسى ۋە سىز Plotly نىڭ نېمە ئۈچۈنلىكىنى ئويلىشىڭىز مۇمكىن. بۇ يەردە سەۋەبى:

  • Plotly نىڭ چوڭايتىلغان كۈتۈك قورالى بار، بۇ بىزگە نۇرغۇن سانلىق مەلۇمات نۇقتىلىرىدىكى ھەر قانداق نورمالسىزلىق ياكى گەپسەتنى بايقاشتا يول قويىدۇ.
  • ئۇ تېخىمۇ كۆپ ماسلاشتۇرۇشنى يول قويىدۇ.
  • ئۇ گرافىكىنى كۆرۈنۈشتە تېخىمۇ جەلپكار قىلىدۇ.

ئورنىتىش

Plotly نى ئورنىتىش

تارقاق نۇقتا گرافىكى (Scatter Plot)

Plotly دىكى تارقاق نۇقتا گرافىكى plotly.express نىڭ scatter() ئۇسۇلى ئارقىلىق قۇرۇلالايدۇ. Seaborn غا ئوخشاش، بۇ يەردىمۇ قوشۇمچە data دەلىلى تەلەپ قىلىنىدۇ.

تارقاق نۇقتا گرافىكى 
import plotly.express as px
import pandas as pd

# سانلىق مەلۇمات بازىسىنى ئوقۇش
data = pd.read_csv("tips.csv")

# تارقاق نۇقتا گرافىكىنى سىزىش
fig = px.scatter(data, x="day", y="tip", color='sex')

ئىنتېراكتىپلىق قوشۇش

Bokeh غا ئوخشاش، plotly مۇ تۈرلۈك ئىنتېراكتىپلىقلارنى تەمىنلەيدۇ. بىز ئۇلارنىڭ بەزىلىرىنى مۇزاكىرە قىلايلى.

تۆۋەنگە تاشلاش تىزىملىكىنى قۇرۇش

تۆۋەنگە تاشلاش تىزىملىكى ھەمىشە ئېكراندا كۆرۈنىدىغان تىزىملىك توپچىسىنىڭ بىر قىسمىدۇر. ھەر بىر تىزىملىك توپچىسى چېكىلگەندە شۇ تىزىملىك توپچىسى ئۈچۈن تاللاشلارنى كۆرسىتەلەيدىغان Menu ۋىدجېتى بىلەن باغلىنىشلىق. Plotly دىكى گرافىكىلارنى ئۆزگەرتىش ئۈچۈن updatemenu ئۇسۇلى بىلەن 4 خىل مۇمكىن ئۇسۇل بار.

تۆۋەنگە تاشلاش تىزىملىكى 
import plotly.graph_objects as px
import pandas as pd

# سانلىق مەلۇمات بازىسىنى ئوقۇش
data = pd.read_csv("tips.csv")


plot = px.Figure(data=[px.Scatter(
    x=data['day'],
    y=data['tip'],
    mode='markers',)
])

# تۆۋەنگە تاشلاش تىزىملىكىنى قوشۇش
plot.update_layout(
    updatemenus=[
        dict(
            buttons=list([
                dict(
                    args=["type", "scatter"],
                    label="تارقاق نۇقتا گرافىكى",
                    method="restyle"
                ),
                dict(
                    args=["type", "bar"],
                    label="ستون گرافىكى",
                    method="restyle"
                )
            ]),
            direction="down",
        ),
    ]
)

plot.show()

سىزغۇچ ۋە تاللىغۇچلارنى قۇرۇش

Plotly دىكى دائىرە سىزغۇچى ئىختىيارىي دائىرە تىپىدىكى كىرگۈزۈش كونتروللىرىدۇر. ئۇ بەلگىلەنگەن ئەڭ تۆۋەن ۋە ئەڭ يۇقىرى دائىرە ئارىسىدىكى بىر قىممەت ياكى دائىرە قىممەتلىرىنى تاللاشقا يول قويىدۇ. دائىرە تاللىغۇچى بولسا گرافىكىدا كۆرسىتىش ئۈچۈن دائىرە تاللاش ئۈچۈن قورال. ئۇ گرافىكىدا ئالدىن تەڭشەلگەن دائىرىلەرنى تاللاش ئۈچۈن توپچىلارنى تەمىنلەيدۇ. ئۇ ئەڭ تۆۋەن ۋە ئەڭ يۇقىرى چېسلا ھەم قولدا كىرگۈزۈش رامكىلىرىنىمۇ تەمىنلەيدۇ.

سىزغۇچ ۋە تاللىغۇچلار 
import plotly.graph_objects as px
import pandas as pd

# سانلىق مەلۇمات بازىسىنى ئوقۇش
data = pd.read_csv("tips.csv")

plot = px.Figure(data=[px.Scatter(
    y=data['tip'],
    mode='lines',)
])

plot.update_layout(
    xaxis=dict(
        rangeselector=dict(
            buttons=list([
                dict(count=1,
                    step="day",
                    stepmode="backward"),
            ])
        ),
        rangeslider=dict(
            visible=True
        ),
    )
)

plot.show()

كۈتۇپخانىلارنى سېلىشتۇرۇش

كۈتۇپخانا ئالاھىدىلىكى ياخشى تەرەپلىرى يامان تەرەپلىرى ئىشلىتىش ئورنى
Matplotlib تۆۋەن سەۋىيە، كۆپ ئىشلىنىشچانلىق كۆپ ماسلاشتۇرۇش، كەڭ قوللىنىش كۆپ كود يېزىش كېرەك، ئىنتېراكتىپلىق تۆۋەن ئادەتتىكى گرافىكىلار، ئىلمىي تەتقىقات
Seaborn يۇقىرى سەۋىيە، ماتپلوتلىب ئاساسىدا ئاز كود بىلەن ياخشى نەتىجە، گۈزەل دىزايىن ماسلاشتۇرۇش چەكلىك، ماتپلوتلىب كېرەك ستاتىستىكىلىق گرافىكىلار، سانلىق مەلۇمات تەھلىلى
Bokeh ئىنتېراكتىپ، تور ئاساسىدىكى يۇقىرى ئىنتېراكتىپلىق، تور كۆرسىتىش ئۆگىنىش قىيىن، كۆپ كود كېرەك تور ئەپلىرى، دېشبوردلار
Plotly ئىنتېراكتىپ، يۇقىرى سەۋىيە يۇقىرى ئىنتېراكتىپلىق، ئاز كود، گۈزەل گرافىكىلار ئېچىۋېتىش مۇرەككەپ، چوڭ ھەجىملىك دېشبوردلار، تور ئەپلىرى، سانلىق مەلۇمات تەھلىلى
