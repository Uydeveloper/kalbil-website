دەرس بايانى

بۇ بابتا سانلىق مەلۇماتنى تازىلاش، قۇرۇق مەلۇماتنى تازىلاش، خاتا مەلۇماتنى تۈزىتىش ھەققىدە ئۆگىنىمىز.

قۇرۇق مەلۇماتنى تازىلاش

NaN (قۇرۇق) مەلۇماتنى تازىلاش ياكى تولدۇرۇش:

import pandas as pd
import numpy as np

# قۇرۇق مەلۇمات بار DataFrame
df = pd.DataFrame({'A': [1, 2, np.nan], 'B': [5, np.nan, np.nan], 'C': [1, 2, 3]})

# قۇرۇق مەلۇماتنى چىقىرىش
df_clean = df.dropna()

# قۇرۇق مەلۇماتنى تولدۇرۇش
df_fill = df.fillna(0)

# ئوتتۇرچە قىممەت بىلەن تولدۇرۇش
df_mean = df.fillna(df.mean())

خاتا مەلۇماتنى تۈزىتىش

خاتا مەلۇماتنى ئالماشتۇرۇش:

# خاتا مەلۇماتنى ئالماشتۇرۇش
df['شەھەر'] = df['شەھەر'].replace('ئۈرۈمچى', 'ئۈرۈمچى شەھىرى')

# بىر نەچچە خاتا مەلۇماتنى ئالماشتۇرۇش
df['شەھەر'] = df['شەھەر'].replace({'ئۈرۈمچى': 'ئۈرۈمچى شەھىرى', 'قەشقەر': 'قەشقەر شەھىرى'})

مەلۇماتنى تازىلاش

شەرت بويىچە مەلۇماتنى تازىلاش:

# ياشى 0 دىن كىچىك بولغانلارنى چىقىرىش
df = df[df['ياشى'] > 0]

# شەھەر ئۇچۇرى يوقلارنى چىقىرىش
df = df[df['شەھەر'].notna()]

چۈشەندۈرۈش

بۇ بابتا قۇرۇق مەلۇماتنى تازىلاش، خاتا مەلۇماتنى تۈزىتىش، شەرت بويىچە مەلۇماتنى تازىلاشنى ئۆگىنىدۇق. بۇ سانلىق مەلۇمات تەھلىلىنىڭ ئەڭ مۇھىم قەدەمى.
